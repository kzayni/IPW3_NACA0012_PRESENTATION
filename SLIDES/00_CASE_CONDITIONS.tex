\begin{frame}{\small NACA0012 -- Modeling Choices Across Submissions}
\scriptsize

\begin{itemize}

    \item \textbf{Turbulence:}
    $k$--$\omega$ formulations were most common (\textbf{62.5\%}), followed by SA (\textbf{37.5\%}).

    \vspace{0.10cm}

    \item \textbf{Droplet formulation:}
    Predominantly Eulerian (\textbf{75\%}), with \textbf{25\%} Lagrangian.

    \vspace{0.10cm}

    \item \textbf{Heat transfer:}
    Documented approaches included recovery-temperature and boundary-layer integral methods.

    \vspace{0.10cm}

    \item \textbf{Thermodynamics:}
    Messinger-based models were dominant (\textbf{71.4\%}); SWIM-based and fluid-film approaches were also used.

    \vspace{0.10cm}

    \item \textbf{Surface roughness:}
    Documented submissions used constant roughness heights ($k_s \approx 0.5$ mm).

    \vspace{0.10cm}

    \item \textbf{Geometry evolution:}
    Most commonly based on Lagrangian/IB/node displacement; level-set, ALE morphing, and interpolation-based approaches were also reported.

    \vspace{0.10cm}

    \item \textbf{Ice accretion strategy:}
    Both single-shot (\textbf{5 submissions}) and multilayer/multishot (\textbf{4 submissions}) approaches were represented.

\end{itemize}

\end{frame}
