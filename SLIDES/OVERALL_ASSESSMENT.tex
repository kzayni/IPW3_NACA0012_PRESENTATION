\section[Assessment]{\textbf{NACA0012 Overall Assessment}}


% ============================================================
% NACA0012 -- Overall Assessment
% ============================================================

\begin{frame}[shrink]{\small NACA0012 -- Overall Assessment}
\scriptsize

\begin{block}{\scriptsize Aerodynamics}
\begin{itemize}
    \item Most submissions exhibit limited grid sensitivity in the integrated aerodynamic coefficients, with $C_D$ showing greater sensitivity than $C_L$ and $C_M$.
    \item L1 $C_p$ distributions are closely clustered and generally agree well with the experimental measurements.
\end{itemize}
\end{block}

\vspace{0.15cm}

\begin{block}{\scriptsize Droplet Impingement}
\begin{itemize}
    \item Collection-efficiency metrics and integrated water mass exhibit comparatively weak spatial grid sensitivity for most submissions.
    \item The two icing conditions produce similar water impingement, while reducing the droplet distribution to a single bin introduces a clearer systematic effect.
\end{itemize}
\end{block}

\vspace{0.15cm}

\begin{block}{\scriptsize Convective Heat Transfer}
\begin{itemize}
    \item Reconstructed recovery-temperature distributions are closely clustered across submissions.
    \item Larger code-to-code differences occur in $HTC$ and consequently in the integrated convective heat transfer.
\end{itemize}
\end{block}

\vspace{0.15cm}

\begin{block}{\scriptsize Ice Accretion}
\begin{itemize}
    \item The first systematic distinction between AE3932 and AE3933 emerges in the resulting ice-accretion response.
    \item Mass-partition and geometric metrics provide additional information beyond total ice mass for interpreting the final ice shapes.
\end{itemize}
\end{block}

\end{frame}
